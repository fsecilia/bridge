cmake_minimum_required(VERSION 3.31.6)
project(BridgeSDLRequirementFixture LANGUAGES NONE)

include("${BRIDGE_SOURCE_DIR}/cmake/BridgeSDL.cmake")
if(NOT BRIDGE_SDL_CASE STREQUAL "missing-canon")
    include("${BRIDGE_SOURCE_DIR}/test/canon/load.cmake")
endif()

if(NOT BRIDGE_SDL_CASE MATCHES "^(missing-provider|missing-vendored|android-installed|android-missing-source)$")
    # The test double models an already-provided SDL target with an explicit version. This test exercises Bridge's
    # requirement policy; actual Canon package selection is tested separately by the dependency integration tests.
    add_library(sdl SHARED IMPORTED)
    add_library(SDL3::SDL3 ALIAS sdl)
    set_property(TARGET sdl PROPERTY SDL_FULL_VERSION "3.4.18")
endif()

if(BRIDGE_SDL_CASE STREQUAL "android-installed" OR BRIDGE_SDL_CASE STREQUAL "android-missing-source")
    set(ANDROID TRUE)
endif()

if(BRIDGE_SDL_CASE STREQUAL "valid")
    bridge_require_sdl(VERSION "3.2.0...<4.0")
    bridge_require_sdl(VERSION "3.4.0...<3.5.0")
elseif(BRIDGE_SDL_CASE STREQUAL "too-new")
    bridge_require_sdl(VERSION "3.5.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "upper-exclusive")
    bridge_require_sdl(VERSION "3.2.0...<3.4.18")
elseif(BRIDGE_SDL_CASE STREQUAL "changed-source")
    bridge_require_sdl(VERSION "3.2.0...<4.0")
    bridge_require_sdl(VERSION "3.2.0...<4.0" EXTERNAL_NAME another-sdl)
elseif(BRIDGE_SDL_CASE STREQUAL "invalid-provider")
    set(BRIDGE_SDL_PROVIDER unspecified CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "invalid-runtime")
    set(BRIDGE_SDL_RUNTIME unspecified CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "missing-canon")
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "missing-provider")
    set(BRIDGE_SDL_PROVIDER installed CACHE STRING "" FORCE)
    set(CMAKE_DISABLE_FIND_PACKAGE_SDL3 TRUE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "preprovided-installed")
    set(BRIDGE_SDL_PROVIDER installed CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "missing-vendored")
    set(BRIDGE_SDL_PROVIDER vendored CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "android-installed")
    set(BRIDGE_SDL_PROVIDER installed CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
elseif(BRIDGE_SDL_CASE STREQUAL "android-missing-source")
    set(BRIDGE_SDL_PROVIDER vendored CACHE STRING "" FORCE)
    bridge_require_sdl(VERSION "3.2.0...<4.0")
else()
    message(FATAL_ERROR "Unknown SDL requirement fixture: ${BRIDGE_SDL_CASE}")
endif()
